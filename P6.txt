import torch, torchvision, matplotlib.pyplot as plt
from PIL import Image
fcn=torchvision.models.segmentation.fcn_resnet50(pretrained=True).eval()
frcnn=torchvision.models.detection.fasterrcnn_resnet50_fpn(pretrained=True).eval()
tfm=torchvision.transforms.ToTensor()


for p in ["road.jpg","road1.jpeg","room.jpg"]:
    img=Image.open(p); t=tfm(img)
    with torch.no_grad():
        seg=fcn(t[None])['out'].argmax(1).squeeze()
        det=frcnn([t])[0]
    fig,ax=plt.subplots(1,3,figsize=(15,5))
    ax[0].imshow(img);ax[1].imshow(seg);ax[2].imshow(img)
    [ax[2].add_patch(plt.Rectangle((x1,y1),x2-x1,y2-y1,fill=False,color='r')) for x1,y1,x2,y2 in det['boxes']]
    [a.axis("off") for a in ax];plt.show()
